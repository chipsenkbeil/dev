-- Lint and test on a linux container with dev lint and dev test when the Lua commands are present
--
--   ci run --local
--   ci run --local --job test

local LINUX = { target = "linux-x86-dev" }

return {
    jobs = {
        lint = {
            matrix = { LINUX },
            run = function()
                ci.sh("if [ -f .dev/lint.lua ]; then dev lint </dev/null; fi")
            end,
        },
        test = {
            matrix = { LINUX },
            run = function()
                ci.sh("if [ -f .dev/test.lua ]; then dev test </dev/null; fi")
            end,
        },
    },
}
