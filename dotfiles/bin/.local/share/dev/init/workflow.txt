name: ci

on:
  pull_request:
  push:
    branches: [@TRUNK@]
    tags: ["*"]

jobs:
  check:
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, windows-latest]
    steps:
      - uses: actions/checkout@v4
      - name: Install nightly rustfmt
        run: rustup toolchain install nightly --profile minimal --component rustfmt
      - uses: taiki-e/install-action@v2
        with:
          tool: cargo-nextest
      - name: Lint
        run: dev lint
        shell: bash
      - name: Test
        run: dev test
        shell: bash
